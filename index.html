Programa de exemplo

Mini sistema de carrinho (lista + dicionário + funções).

Mini Carrinho em Python
def adicionar_item(carrinho, nome, preco):
    carrinho.append({"nome": nome, "preco": preco})

def total(carrinho):
    return sum(item["preco"] for item in carrinho)

def resumo(carrinho):
    if not carrinho:
        return "Carrinho vazio."
    linhas = []
    for i, item in enumerate(carrinho, start=1):
        linhas.append(f"{i}. {item['nome']} — R$ {item['preco']:.2f}")
    linhas.append(f"TOTAL: R$ {total(carrinho):.2f}")
    return "\n".join(linhas)

def main():
    carrinho = []
    adicionar_item(carrinho, "Laço Rosa Bebê", 19.90)
    adicionar_item(carrinho, "Kit Laços (3 un.)", 39.90)
    adicionar_item(carrinho, "Presilha Glitter", 9.90)

    print("=== RESUMO DO PEDIDO ===")
    print(resumo(carrinho))

if __name__ == "__main__":
    main()
